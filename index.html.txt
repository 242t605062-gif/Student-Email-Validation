    <script>
        const form = document.getElementById("emailForm");
        const emailInput = document.getElementById("email");
        const result = document.getElementById("result");

        const DOMAIN = "@ashokacollege.in";

        // Allowed characters in the email username.
        function isUsernameCharacter(char) {
            return /^[a-zA-Z0-9._%+-]$/.test(char);
        }

        // Simulate the DFA.
        function validateEmailDFA(email) {
            const path = ["q0"];
            let state = "q0";

            // Empty input is rejected.
            if (email.length === 0) {
                return {
                    accepted: false,
                    path: ["q0", "DEAD"],
                    reason: "Email address cannot be empty."
                };
            }

            // First character must begin a username.
            if (!/^[a-zA-Z0-9_%+-]$/.test(email[0])) {
                return {
                    accepted: false,
                    path: ["q0", "DEAD"],
                    reason: "The email must begin with a valid username character."
                };
            }

            state = "q1";
            path.push(state);

            let i = 1;

            // q1: consume the username.
            while (i < email.length &&
                   email[i] !== "@") {

                if (!isUsernameCharacter(email[i])) {
                    path.push("DEAD");

                    return {
                        accepted: false,
                        path,
                        reason: "The username contains an invalid character."
                    };
                }

                state = "q1";
                path.push(state);
                i++;
            }

            // Username must be followed by @.
            if (i >= email.length ||
                email[i] !== "@") {
                path.push("DEAD");

                return {
                    accepted: false,
                    path,
                    reason: "The email must contain @ before the domain."
                };
            }

            // Basic username dot checks.
            const username = email.slice(0, i);

            if (
                username.endsWith(".") ||
                username.includes("..")
            ) {
                path.push("DEAD");

                return {
                    accepted: false,
                    path,
                    reason: "The username cannot end with a dot or contain consecutive dots."
                };
            }

            // q2 onward: match the exact required domain.
            state = "q2";
            path.push(state);
            i++;

            const domain = DOMAIN.slice(1);

            for (let j = 0; j < domain.length; j++) {
                if (
                    i >= email.length ||
                    email[i].toLowerCase() !== domain[j]
                ) {
                    path.push("DEAD");

                    return {
                        accepted: false,
                        path,
                        reason:
                            "The domain must be exactly ashokacollege.in."
                    };
                }

                state = "q" + (j + 3);
                path.push(state);
                i++;
            }

            // No extra characters may follow the domain.
            if (i !== email.length) {
                path.push("DEAD");

                return {
                    accepted: false,
                    path,
                    reason:
                        "Extra characters after the college domain are not allowed."
                };
            }

            // Accept only in the final state.
            return {
                accepted: true,
                path,
                reason:
                    "The email matches the required college domain."
            };
        }

        form.addEventListener("submit", function(event) {
            event.preventDefault();

            const email = emailInput.value.trim();
            const answer = validateEmailDFA(email);

            result.className = "result " +
                (answer.accepted ? "accepted" : "rejected");

            result.innerHTML = `
                <h3>Validation Result</h3>

                <p class="status">
                    ${answer.accepted ? "✓ Email is ACCEPTED" :
                      "✗ Email is REJECTED"}
                </p>

                <p>
                    <strong>Email:</strong>
                    ${escapeHTML(email)}
                </p>

                <p>
                    <strong>Reason:</strong>
                    ${answer.reason}
                </p>

                <div class="dfa-path">
                    <h4>DFA State Transition Path</h4>
                    <p class="path-text">
                        ${answer.path.join(" → ")}
                    </p>
                </div>

                <p style="margin-top: 15px;">
                    <strong>Final Result:</strong>
                    ${answer.accepted ?
                      "Accepted by the DFA (final state)." :
                      "Rejected by the DFA (dead state)."}
                </p>
            `;
        });

        // Prevent user input from being interpreted as HTML.
        function escapeHTML(value) {
            return value.replace(/[&<>"']/g, function(char) {
                const entities = {
                    "&": "&amp;",
                    "<": "&lt;",
                    ">": "&gt;",
                    '"': "&quot;",
                    "'": "&#39;"
                };

                return entities[char];
            });
        }
